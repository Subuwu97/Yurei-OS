cmake_minimum_required(VERSION 3.16)
project(yurei-install-gui VERSION 0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets)
add_executable(yurei-install-gui main.cpp artwork.qrc)
target_link_libraries(yurei-install-gui PRIVATE Qt6::Widgets)
install(TARGETS yurei-install-gui RUNTIME DESTINATION bin)
install(FILES yurei-install-gui.desktop DESTINATION share/applications)
install(FILES ARTWORK.md DESTINATION share/licenses/yurei-install-gui)
include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_executable(installer-model-tests model_tests.cpp)
    target_link_libraries(installer-model-tests PRIVATE Qt6::Test Qt6::Widgets)
    add_test(NAME installer-model COMMAND installer-model-tests)
    set_tests_properties(installer-model PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
